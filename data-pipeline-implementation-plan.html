TL;DR

  1. Mirror raw data into data/raw/de/ (16 JSON) + data/raw/wiki/ (15 Lua) — checked into git.
  2. Build script reads from disk, parses Lua with luaparse, merges with curated YAML, emits the same items-index.json shape we have today.
  3. Three new fields on weapons: displayClass (wiki Class), modPool (= displayClass + small override map), compatTags (wiki CompatibilityTags).
  4. Mod-compat predicate collapses from ~80 LOC of guessing to two structural checks.
  5. Build never hits the network. The mirror IS the pin. Git diff is the wiki changelog.

Phase 1 · scripts/sync-de.ts 
import { decode as lzmaDecode } from "lzma";
import { writeFileSync } from "node:fs";
import { createHash } from "node:crypto";

const INDEX_URL = "https://content.warframe.com/PublicExport/index_en.txt.lzma";
const BLOB_URL = (h: string) =>
  `https://content.warframe.com/PublicExport/Lzma/${h}`;

const indexBytes = new Uint8Array(await (await fetch(INDEX_URL)).arrayBuffer());
const indexText  = new TextDecoder().decode(lzmaDecode(indexBytes));
// "ExportWeapons_en.json!00_aBcD...\nExportUpgrades_en.json!00_..."

const entries = indexText.trim().split("\n").map(line => {
  const [name, hash] = line.split("!");
  return { name, hash };
});

for (const { name, hash } of entries) {
  const compressed = new Uint8Array(await (await fetch(BLOB_URL(hash))).arrayBuffer());
  const raw   = new TextDecoder().decode(lzmaDecode(compressed));
  const clean = raw.replace(/\x00/g, "");   // sanitize NUL bytes
  const parsed = JSON.parse(clean);            // validate
  writeFileSync(`data/raw/de/${name}`, JSON.stringify(parsed, null, 2));
}

writeFileSync("data/PINS.json", JSON.stringify({
  deIndexHash: createHash("sha256").update(indexText).digest("hex"),
  fetchedAt:   new Date().toISOString(),
  entries,
}, null, 2));

Phase 2 · scripts/build/read-wiki.ts (luaparse glue) 
import { parse as luaparse } from "luaparse";
import { readFileSync } from "node:fs";

export function readWikiModule(path: string): Record<string, unknown> {
  const src = readFileSync(path, "utf-8");
  const ast = luaparse(src, { comments: false });
  const ret = ast.body.find((n: any) => n.type === "ReturnStatement");
  if (!ret) throw new Error(`No return in ${path}`);
  return luaTableToJs(ret.arguments[0]);
}

function luaTableToJs(node: any): any {
  if (node.type === "TableConstructorExpression") {
    const isArray = node.fields.every((f: any) => f.type === "TableValue");
    if (isArray) return node.fields.map((f: any) => luaTableToJs(f.value));
    const obj: Record<string, any> = {};
    for (const f of node.fields) {
      const key = f.type === "TableKey"
        ? luaTableToJs(f.key)         // ["Coda Bubonico"] = {...}
        : f.key.name;                  // Hek = {...}
      obj[key] = luaTableToJs(f.value);
    }
    return obj;
  }
  if (node.type === "StringLiteral")  return node.value;
  if (node.type === "NumericLiteral") return node.value;
  if (node.type === "BooleanLiteral") return node.value;
  if (node.type === "NilLiteral")     return null;
  if (node.type === "UnaryExpression" && node.operator === "-")
    return -luaTableToJs(node.argument);
  throw new Error(`Unhandled Lua node: ${node.type}`);
}

Phase 4 · build/merge-weapons.ts (the fail-loud core) 
const KNOWN_PRODUCT_CATEGORIES = new Set([
  "Pistols", "Melee", "LongGuns", "SpecialItems",
  "SentinelWeapons", "SpaceGuns", "SpaceMelee", "OperatorAmps",
]);

const KNOWN_MOD_POOLS = new Set([
  "Pistol", "Rifle", "Shotgun", "Melee", "Sniper", "Bow",
  "Arch-Gun", "Arch-Melee", "Parazon", "Necramech",
  "Plexus", "Tome", "Robotic", "Sentinel", "Amp", /* ... */
]);

const KNOWN_WIKI_CLASSES = new Set([/* enumerated in Phase 0 from raw .lua */]);

function mergeWeapon(deWeapon, wikiWeapon, ours) {
  if (!KNOWN_PRODUCT_CATEGORIES.has(deWeapon.productCategory))
    throw new Error(`Unknown productCategory: ${deWeapon.productCategory} on ${deWeapon.name}`);
  if (wikiWeapon && !KNOWN_WIKI_CLASSES.has(wikiWeapon.Class))
    throw new Error(`Unknown wiki Class: ${wikiWeapon.Class} on ${deWeapon.name}`);

  const displayClass = wikiWeapon?.Class
    ?? ours.stubs[deWeapon.uniqueName]?.displayClass;
  const modPool = ours.modPools[deWeapon.name] ?? displayClass;
  if (!KNOWN_MOD_POOLS.has(modPool))
    throw new Error(`Unknown modPool: ${modPool} (from ${deWeapon.name}). Add override.`);

  return {
    uniqueName: deWeapon.uniqueName,
    name:       deWeapon.name,
    displayClass,
    modPool,
    compatTags:     wikiWeapon?.CompatibilityTags ?? [],
    polarities:     wikiWeapon?.Polarities ?? [],
    exilusPolarity: wikiWeapon?.ExilusPolarity ?? null,
    family:         wikiWeapon?.Family ?? null,
    slot:           wikiWeapon?.Slot ?? null,
    traits:         wikiWeapon?.Traits ?? [],
    masteryReq:     deWeapon.masteryReq,
    damagePerShot:  deWeapon.damagePerShot,   // DE wins for raw numbers
    fireRate:       deWeapon.fireRate,
    magazineSize:   deWeapon.magazineSize,
    // ... rest of DE stats
  };
}

Hard rules

  • Use uv run python for any one-off scripts. Plain python/python3 is blocked.
  • Don't add deps without asking — except luaparse and lzma (already approved by this plan).
  • bun run typecheck is the source of truth — vite build doesn't catch web/shared type errors.
  • apps/web/public/data/items-index.json is checked into the repo — regenerate with bun scripts/build-items-index.ts after any build-script change.
  • Don't modify apps/web/src/components/ui/ — override via className.
  • Wiki content is CC-BY-SA — leave ATTRIBUTION.md in place, preserve source-URL comments at the top of each .lua.
  • If a wiki module's Lua shocks luaparse: don't sed-patch the mirror. File a wiki ticket or add a curated override. The mirror IS the source of truth.
  • KNOWN_* assertion sets are load-bearing. Don't soften any of them to warnings. The whole point is fail-loud on drift.
  • Don't assume Warframe game-mechanic facts from memory (CLAUDE.md rule). Verify against the wiki or data files before encoding.
